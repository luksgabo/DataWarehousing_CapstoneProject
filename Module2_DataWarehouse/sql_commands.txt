## Task 5
SELECT
    c.country,
    t.category,
    SUM(f.amount) AS totalsales
FROM
    "FactSales" f
INNER JOIN
    "DimCountry" c ON f.countryid = c.countryid
INNER JOIN
    "DimCategory" t ON f.categoryid = t.categoryid
GROUP BY GROUPING SETS (
    (c.country, t.category),
    c.country,
    t.category,
    ()
)
ORDER BY
    c.country,
    t.category;

CREATE MATERIALIZED VIEW total_sales_per_country (country, totalsales) AS
(SELECT c.country, sum(f.amount) from "FactSales" f
LEFT JOIN "DimCountry" c ON f.countryid = c.countryid
group by c.country);
